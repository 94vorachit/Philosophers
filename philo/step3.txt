## Small next step: thread startup scaffold

This is the next tiny step after initialization: create threads, wait on them, and keep the philosopher body as a placeholder. No eating/death logic yet.

Add this to philo.h:

```c
struct s_sim
{
	int				philo_count;
	int				time_to_die;
	int				time_to_eat;
	int				time_to_sleep;
	int				must_eat;
	long long		start_time_ms;
	bool			stop;
	pthread_mutex_t	stop_lock;
	pthread_mutex_t	print_lock;
	pthread_mutex_t	state_lock;
	pthread_mutex_t	*forks;
	t_philo			*philos;
	pthread_t		*threads;
};

int				start_simulation(t_sim *sim);
```

Create a new file or add this to a small new runtime file, for example philo/srcs/simulation.c:

```c
#include "philo.h"

static bool	is_stopped(t_sim *sim)
{
	bool	stopped;

	pthread_mutex_lock(&sim->stop_lock);
	stopped = sim->stop;
	pthread_mutex_unlock(&sim->stop_lock);
	return (stopped);
}

static void	*philo_routine(void *arg)
{
	t_philo	*philo;

	philo = (t_philo *)arg;
	while (!is_stopped(philo->sim))
		usleep(1000);
	return (NULL);
}

int	start_simulation(t_sim *sim)
{
	int	index;

	sim->threads = malloc(sizeof(pthread_t) * sim->philo_count);
	if (!sim->threads)
		return (0);
	index = 0;
	while (index < sim->philo_count)
	{
		if (pthread_create(&sim->threads[index], NULL, philo_routine,
				&sim->philos[index]) != 0)
			return (0);
		index++;
	}
	index = 0;
	while (index < sim->philo_count)
	{
		pthread_join(sim->threads[index], NULL);
		index++;
	}
	free(sim->threads);
	sim->threads = NULL;
	return (1);
}
```

Then in main.c, after `init_sim`, call:

```c
if (!start_simulation(&sim))
{
	print_error("Error: could not start simulation\n");
	return (1);
}
```

This is intentionally minimal and safe:
- no real eating logic
- no death monitor
- no fork locking yet

It just proves the runtime scaffold is working.
