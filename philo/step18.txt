void	handle_death(t_sim *sim, int index)
{
	pthread_mutex_lock(&sim->state_lock);
	if (now_ms() - sim->philos[index].last_meal_ms <= sim->time_to_die)
	{
		pthread_mutex_unlock(&sim->state_lock);
		return ;

	printf("%lld %d died\n",

	pthread_mutex_unlock(&sim->print_lock);

	set_stop(sim);
}
