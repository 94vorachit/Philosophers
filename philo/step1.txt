## Step 1 — freeze the behavior and define the core model

This is the first clean milestone: strict argument validation + a shared simulation state model. Nothing thread-related yet.

```c
#ifndef PHILO_H
# define PHILO_H

# include <errno.h>
# include <limits.h>
# include <pthread.h>
# include <stdbool.h>
# include <stdio.h>
# include <stdlib.h>
# include <string.h>
# include <sys/time.h>
# include <unistd.h>

# define PHILO_MAX_COUNT 200

typedef struct s_sim t_sim;

typedef struct s_philo
{
	int				id;
	int				left_fork;
	int				right_fork;
	long long		last_meal_ms;
	int				meals_eaten;
	pthread_mutex_t	*left_lock;
	pthread_mutex_t	*right_lock;
	t_sim			*sim;
}	t_philo;

struct s_sim
{
	int				philo_count;
	int				time_to_die;
	int				time_to_eat;
	int				time_to_sleep;
	int				must_eat;
	long long		start_time_ms;
	bool			stop;
	pthread_mutex_t	stop_lock;
	pthread_mutex_t	print_lock;
	pthread_mutex_t	state_lock;
	pthread_mutex_t	*forks;
	t_philo			*philos;
};

int		parse_args(int ac, char **av, t_sim *sim);
void	print_error(const char *msg);
long long	now_ms(void);

#endif
```

```c
#include "philo.h"

static int	is_number(const char *s)
{
	int	i;

	if (!s || !*s)
		return (0);
	i = 0;
	if (s[i] == '+' || s[i] == '-')
		i++;
	if (!s[i])
		return (0);
	while (s[i])
	{
		if (s[i] < '0' || s[i] > '9')
			return (0);
		i++;
	}
	return (1);
}

static int	get_int_value(const char *s, int *out)
{
	char	*end;
	long	value;

	if (!is_number(s))
		return (0);
	errno = 0;
	value = strtol(s, &end, 10);
	if (errno != 0 || *end != '\0')
		return (0);
	if (value < INT_MIN || value > INT_MAX)
		return (0);
	*out = (int)value;
	return (1);
}

int	parse_args(int ac, char **av, t_sim *sim)
{
	if (ac < 5 || ac > 6)
		return (0);
	if (!get_int_value(av[1], &sim->philo_count))
		return (0);
	if (!get_int_value(av[2], &sim->time_to_die))
		return (0);
	if (!get_int_value(av[3], &sim->time_to_eat))
		return (0);
	if (!get_int_value(av[4], &sim->time_to_sleep))
		return (0);
	if (sim->philo_count < 1 || sim->philo_count > PHILO_MAX_COUNT)
		return (0);
	if (sim->time_to_die <= 0 || sim->time_to_eat <= 0 || sim->time_to_sleep <= 0)
		return (0);
	sim->must_eat = -1;
	if (ac == 6)
	{
		if (!get_int_value(av[5], &sim->must_eat))
			return (0);
		if (sim->must_eat < 0)
			return (0);
	}
	return (1);
}
```

```c
#include "philo.h"

void	print_error(const char *msg)
{
	if (msg)
		write(2, msg, strlen(msg));
}

long long	now_ms(void)
{
	struct timeval	tv;

	gettimeofday(&tv, NULL);
	return ((long long)tv.tv_sec * 1000LL + (long long)tv.tv_usec / 1000LL);
}

int	main(int ac, char **av)
{
	t_sim	sim;

	if (!parse_args(ac, av, &sim))
	{
		print_error("Error: invalid arguments\n");
		return (1);
	}
	/* init phase comes next */
	(void)sim;
	return (0);
}
```

### Why this is the correct first step
- It locks the required CLI contract before writing the simulation logic.
- It defines the shared state that the later thread routines will use.
- It prevents the most common early bug: accepting invalid or overflowed input.
